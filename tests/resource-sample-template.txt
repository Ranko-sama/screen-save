Developer measurement template used for the twelve comparative samples.
Replace MODE with off / normal / gamma / matrix and ROUND with 1 / 2 / 3.
Run in PowerShell from the project directory while the matching browser round runs.
The temporary keep-awake request belongs to this 30-s comparison helper only.
It is NOT used during the 15-minute soak, where only ScreenSave requests keep-awake.

Add-Type -TypeDefinition 'using System.Runtime.InteropServices; public static class BenchAwake { [DllImport("kernel32.dll")] public static extern uint SetThreadExecutionState(uint flags); }'; [void][BenchAwake]::SetThreadExecutionState(2147483651); $p=Get-Process ScreenSave -ErrorAction SilentlyContinue; $dwmPid=(Get-Process dwm).Id; $start=Get-Date; $cpuBefore=if($p){$p.CPU}else{0}; $c=Get-Counter '\GPU Engine(*)\Utilization Percentage','\Process(dwm)\% Processor Time','\Process(dwm)\Working Set - Private' -SampleInterval 1 -MaxSamples 30; $seconds=((Get-Date)-$start).TotalSeconds; $appCpu=0; $appPrivate=0; $appWorking=0; if($p){$p.Refresh();$appCpu=100*($p.CPU-$cpuBefore)/$seconds;$appPrivate=$p.PrivateMemorySize64;$appWorking=$p.WorkingSet64}; $gpu=@($c.CounterSamples | Where-Object InstanceName -Like "pid_$($dwmPid)_*" | Group-Object Timestamp | ForEach-Object {($_.Group | Measure-Object CookedValue -Sum).Sum}); $row=[ordered]@{mode='MODE';round=ROUND;seconds=$seconds;app_cpu_pct=$appCpu;app_private=$appPrivate;app_working=$appWorking;dwm_cpu_pct=($c.CounterSamples | Where-Object Path -Like '*\process(dwm)\% processor time' | Measure-Object CookedValue -Average).Average;dwm_private=($c.CounterSamples | Where-Object Path -Like '*\process(dwm)\working set - private' | Measure-Object CookedValue -Average).Average;dwm_gpu=($gpu | Measure-Object -Average).Average}; $row | ConvertTo-Json -Compress | Add-Content tests/artifacts/gamma-resources-complete.jsonl; $row | ConvertTo-Json; [void][BenchAwake]::SetThreadExecutionState(2147483648)
